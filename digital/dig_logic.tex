\section {Multivariable Theorems}

	\begin{enumerate}[label=(\alph*)]
		\item $x+y=y+x$
		\item $x \cdot y = y \cdot x$
		\item $x + (y + z) = (x + y) + z = x + y + z$
		\item $x(yz) = (xy)z = xyz$
		\item $x(y + z) = xy + xz$
		\item $(w + x)(y + z) = wy + xy + wz + xz$
		\item $x + xy = x$
		\item $x + \overline{x}y = x + y$
		\item $\overline{x} + xy = \overline{x} + y$
	\end{enumerate}

\section {DeMorgan Theorems}

	\begin{enumerate}[label=(\alph*)]
		\item $(\overline{x+y}) = \overline{x} \cdot \overline{y}$
		\item $(\overline{x \cdot y}) = \overline{x} + \overline{y}$
	\end{enumerate}
